# Auto Fund Lobbies - script values
#
# NOTE: the influence thresholds for creating pacts are computed with
# plain variable arithmetic inside afl_weekly_update (scripted_effects) -
# script values that read country variables evaluate to 'none' when used
# through set_variable, so no script value here may reference var:.

# ------------------------------------------------------------------
# How attractive a country is as a funding target.
# this = candidate country, root = the automating country
# ------------------------------------------------------------------
afl_fund_target_score = {
	value = country_rank
	if = {
		limit = {
			any_political_lobby = {
				target = root
				OR = {
					is_political_lobby_type = lobby_pro_country
					is_political_lobby_type = lobby_pro_overlord
				}
			}
		}
		add = 100
	}
	if = {
		limit = { is_subject_of = root }
		add = 50
	}
	if = {
		limit = { is_in_same_power_bloc = root }
		add = 25
	}
}

# Inverse of the target score: the least valuable funded country is
# defunded first when influence drops below the protected reserve.
afl_defund_priority = {
	value = 0
	subtract = afl_fund_target_score
}
