# Auto Fund Lobbies - scripted effects

# Weekly automation tick, fired from je_auto_fund_lobbies' on_weekly_pulse
# and from the "Check Now" journal entry button.
# root = the country owning the journal entry.
# All work happens inside the iterator bodies (vanilla pattern) - scopes
# saved inside an iterator are not read after it closes.
afl_weekly_update = {
	# Country variables do not follow the journal entry through revolutions
	# or tag switches, so restore the default reserve if it is missing.
	if = {
		limit = {
			NOT = { has_variable = afl_min_influence }
		}
		set_variable = {
			name = afl_min_influence
			value = 100
		}
	}

	if = {
		limit = {
			has_variable = afl_enabled
			is_ai = no
			has_dlc_feature = lobbies
		}

		# 1) Forget tracked pacts that already broke on their own
		#    (war, relations, rank loss, default...).
		every_in_list = {
			variable = afl_funded_countries
			limit = {
				NOT = {
					has_diplomatic_pact = {
						who = root
						type = fund_lobbies
						is_initiator = no
					}
				}
			}
			root = {
				remove_list_variable = {
					name = afl_funded_countries
					target = prev
				}
			}
		}

		# 2) Below the protected reserve: cancel the least valuable
		#    automated pact, one per week. Manually created pacts are
		#    never in the list, so they are never touched.
		if = {
			limit = { influence < var:afl_min_influence }
			ordered_in_list = {
				variable = afl_funded_countries
				order_by = afl_defund_priority
				position = 0
				root = {
					remove_diplomatic_pact = {
						country = prev
						type = fund_lobbies
					}
					remove_list_variable = {
						name = afl_funded_countries
						target = prev
					}
				}
			}
		}

		# 3) Comfortably above the reserve: fund lobbies in the best new
		#    target, one pact per week. The pact also transfers money, so
		#    only spend while the treasury is healthy: not defaulted, weekly
		#    balance positive, and debt below 25% of reserves (the same
		#    threshold at which the vanilla AI breaks these pacts).
		else_if = {
			limit = {
				has_technology_researched = central_banking
				NOT = { is_country_type = decentralized }
				is_revolutionary = no
				in_default = no
				scaled_debt < 0.25
				net_fixed_income > 0
			}
			set_variable = {
				name = afl_threshold_gp
				value = afl_pact_threshold_great_power
			}
			set_variable = {
				name = afl_threshold_mp
				value = afl_pact_threshold_major_power
			}
			set_variable = {
				name = afl_threshold_lp
				value = afl_pact_threshold_default
			}
			ordered_country = {
				limit = {
					afl_valid_fund_target = yes
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							root = { influence >= var:afl_threshold_gp }
						}
						AND = {
							country_rank = rank_value:major_power
							root = { influence >= var:afl_threshold_mp }
						}
						AND = {
							country_rank < rank_value:major_power
							root = { influence >= var:afl_threshold_lp }
						}
					}
				}
				order_by = afl_fund_target_score
				position = 0
				root = {
					create_diplomatic_pact = {
						country = prev
						type = fund_lobbies
					}
					add_to_variable_list = {
						name = afl_funded_countries
						target = prev
					}
				}
			}
			remove_variable = afl_threshold_gp
			remove_variable = afl_threshold_mp
			remove_variable = afl_threshold_lp
		}
	}
}
