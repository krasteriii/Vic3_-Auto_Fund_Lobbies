# Auto Fund Lobbies - scripted triggers

# Whether a country is a valid target for a new automated fund_lobbies pact.
# Mirrors the pact's potential + requirement_to_maintain blocks
# (common/diplomatic_actions/30_fund_lobby.txt) so pacts are not created
# just to break the following week.
# this = candidate country, root = the automating country
afl_valid_fund_target = {
	NOT = { root = this }
	NOT = { is_country_type = decentralized }
	is_revolutionary = no
	NOT = {
		has_diplomatic_pact = {
			who = root
			type = fund_lobbies
		}
	}
	NOT = { has_war_with = root }
	country_rank <= root.country_rank
	relations:root > relations_threshold:poor
	root = {
		has_diplomatic_relevance = prev
	}
}
