# Auto Fund Lobbies - settings journal entry
# Added explicitly to human players at campaign start (see
# common/on_actions/afl_on_actions.txt), with a monthly fallback so the
# mod also works when enabled mid-save or after a tag switch.
# Never completes; acts as a permanent settings panel. The weekly
# automation itself only creates pacts once Central Banking is researched
# (the tech that unlocks the Fund Lobbies diplomatic action).

je_auto_fund_lobbies = {
	icon = "gfx/interface/icons/political_lobby_icons/lobby_pro.dds"

	group = je_group_foreign_affairs

	scripted_button = afl_button_enable
	scripted_button = afl_button_disable
	scripted_button = afl_button_run_now
	scripted_button = afl_button_debug_test
	scripted_button = afl_button_reserve_down
	scripted_button = afl_button_reserve_up

	is_shown_when_inactive = {
		is_player = yes
		has_dlc_feature = lobbies
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_variable = afl_min_influence }
			}
			set_variable = {
				name = afl_min_influence
				value = 100
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_created
				trigger = {
					has_variable = afl_enabled
					has_variable = afl_diag_pact_created
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_stuck
				trigger = {
					has_variable = afl_enabled
					has_variable = afl_diag_have_candidate
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_defunded
				trigger = {
					has_variable = afl_enabled
					has_variable = afl_diag_defunded
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_cantafford
				trigger = {
					has_variable = afl_enabled
					has_variable = afl_diag_valid_exists
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_nocandidate
				trigger = {
					has_variable = afl_enabled
					has_variable = afl_diag_money_ok
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_blocked
				trigger = {
					has_variable = afl_enabled
					has_variable = afl_diag_ticked
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_notrun
				trigger = {
					has_variable = afl_enabled
				}
			}
			triggered_desc = {
				desc = je_auto_fund_lobbies_status_disabled
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			afl_weekly_update = yes
		}
	}

	weight = 10
}
